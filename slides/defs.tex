\NewDocumentCommand{\bbox}{o}{%
  \IfNoValueTF{#1}{%
    \coordinate (topright)    at (0.99\textwidth, 0.85\textheight);
    \coordinate (bottomleft)  at (0mm, 0mm);
    \coordinate (topleft)     at (topright -| bottomleft);
    \coordinate (bottomright) at (topright |- bottomleft);
  }{%
    \coordinate (centerpoint) at (current page.center);
    \coordinate (topleft)     at ([xshift=-0.495\textwidth, yshift= 0.425\textheight]centerpoint);
    \coordinate (bottomright) at ([xshift= 0.495\textwidth, yshift=-0.425\textheight]centerpoint);
    \coordinate (topright)    at (bottomright |- topleft);
    \coordinate (bottomleft)  at (topleft |- bottomright);
  }%
  \coordinate (topmid)    at ($(topleft)!0.5!(topright)$);
  \coordinate (bottommid) at (topmid |- bottomright);
  \draw[bbox] (topleft) rectangle (bottomright);
}
